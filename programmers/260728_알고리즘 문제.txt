package programmers;

// 나누어 떨어지는 숫자배열
class Solution {
    public int[] solution(int[] arr, int divisor) {

        // 결과의 배열의 크기를 결정하기 위해 divisor로 나누어 떨어지는 원소의 개수를 먼저 탐색
    
        int count = 0;

        for(int i=0; i<arr.length; i++) {
            if(arr[i] % divisor ==0) {
                count++;
            }
        }

        // 조건을 만족하는 원소가 없다면 -1이 담긴 배열 반환
        if(count == 0) {
            return new int[] {-1};       
        }

        // 앞에서 구한 count를 이용해 결과로 반환할 배열을 생성
        int[] answer =  new int[count];

        // answer 배열에 값을 저장할 위치
        int index = 0;

        for(int i=0;i<arr.length;i++) {
            if(arr[i] % divisor == 0) {
                // 조건을 만족한 현재 원소를 결과 배열 index 위치에 저장
                answer[index] = arr[i];

                index++;
            }
        }

        // 오름차순 정렬
        Arrays.sort(answer);

        return answer;
    }
}
